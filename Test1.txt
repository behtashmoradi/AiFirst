/// <summary>
/// Returns the nth prime number (1-based): NthPrime(1) = 2, NthPrime(2) = 3, NthPrime(5) = 11.
/// Uses a Sieve of Eratosthenes sized by an upper bound on the nth prime,
/// so no per-number trial division is needed.
/// </summary>
public static int NthPrime(int n)
{
    if (n < 1)
        throw new ArgumentOutOfRangeException(nameof(n), "n must be at least 1.");

    // For n >= 6, the nth prime is < n * (ln n + ln ln n) (Rosser's theorem).
    int limit = n < 6
        ? 15
        : (int)(n * (Math.Log(n) + Math.Log(Math.Log(n)))) + 1;

    bool[] isComposite = new bool[limit + 1];
    int count = 0;

    for (int i = 2; i <= limit; i++)
    {
        if (isComposite[i])
            continue;

        count++;
        if (count == n)
            return i;

        // Mark multiples of i, starting at i*i (smaller ones are already marked).
        for (long j = (long)i * i; j <= limit; j += i)
            isComposite[j] = true;
    }

    throw new InvalidOperationException("Sieve limit too small."); // unreachable
}

/// <summary>
/// Checks a single number for primality. Skips multiples of 2 and 3,
/// so only numbers of the form 6k ± 1 are tried as divisors.
/// </summary>
public static bool IsPrime(int number)
{
    if (number < 2) return false;
    if (number < 4) return true;            // 2 and 3
    if (number % 2 == 0 || number % 3 == 0) return false;

    for (long i = 5; i * i <= number; i += 6)
    {
        if (number % i == 0 || number % (i + 2) == 0)
            return false;
    }
    return true;
}
